# The portable host: builds the recompiled game (lifter output) together with the platform layer.
#
#   cmake -S port -B build/port -DPORT_GENERATED=<lifter output> [-DPORT_BACKEND=sdl2|null]
#   cmake --build build/port
#
# PORT_GENERATED contains nd/ (SS2.exe) and mods/ (mods.c), plus ao/ sq/ lv/ fm/ for the recompiled DLLs when they
# were lifted. A port to a new platform adds a directory under backends/ and selects it with PORT_BACKEND.
cmake_minimum_required(VERSION 3.16)
project(ss2port C)
set(PORT_GENERATED "" CACHE PATH "lifter output directory (contains nd/ and mods/)")
set(PORT_BACKEND "sdl2" CACHE STRING "platform backend: sdl2 (window, input, audio, OpenGL) or null (headless)")
if(WIN32)
  set(PORT_SYSTEM "win32" CACHE STRING "system part of the platform layer (memory, threads, files)")
else()
  set(PORT_SYSTEM "posix" CACHE STRING "system part of the platform layer (memory, threads, files)")
endif()
option(PORT_PREEMPT "let other guest threads run at loop heads (needed for games that spin-wait)" ON)
if(NOT PORT_GENERATED)
  message(FATAL_ERROR "set -DPORT_GENERATED=<lifter output directory>")
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
if(NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  message(FATAL_ERROR "the generated code needs GCC or Clang (it uses __builtin_setjmp and friends)")
endif()

# ---- the recompiled game (generated, unchanged between platforms)
file(GLOB GEN_SRC ${PORT_GENERATED}/nd/*.c ${PORT_GENERATED}/ao/*.c ${PORT_GENERATED}/sq/*.c ${PORT_GENERATED}/lv/*.c ${PORT_GENERATED}/fm/*.c ${PORT_GENERATED}/mods/*.c)
set(GEN_DEFS "")
if(PORT_PREEMPT)
  list(APPEND GEN_DEFS RT_PREEMPT)
endif()
option(PORT_PROF "sampling profiler: logs the guest functions and imports that take the most time (core/prof.c; needs PORT_PREEMPT)" OFF)
if(PORT_PROF)
  list(APPEND GEN_DEFS RT_PROF)
endif()
option(PORT_ABI_CHECK "code lifted with --cache-regs: check that indirect calls and imports keep EBX/EBP/ESI/EDI; log (once per call site) and correct violations (runtime/rt_abi_check.c)" OFF)
if(PORT_ABI_CHECK)
  list(APPEND GEN_DEFS RT_ABI_CHECK)
endif()
add_library(game OBJECT ${GEN_SRC})
target_include_directories(game PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../runtime ${PORT_GENERATED}/nd)
target_compile_definitions(game PRIVATE ${GEN_DEFS})
# -ffp-contract=off: x87 math is translated to C doubles (in C locals with --cache-regs); fusing a*b+c on hosts with
# a fused multiply-add would round differently from the original. (The Vita's VFPv3 has none, so it costs nothing there.)
# -fno-math-errno: sqrt and friends need not set the host's errno, which recompiled code never reads (the guest C runtime
# has its own errno in guest memory), so sqrt becomes one instruction without a fallback call.
target_compile_options(game PRIVATE -w -fwrapv -fno-strict-aliasing -ffp-contract=off -fno-math-errno)
# Optimisation level of the generated code. The build type's flags come first and the last -O wins, so these override it.
# PORT_GAME_OPT applies to every generated file (e.g. -O2: -O3, the Release default, unrolls and duplicates code, and the
# generated code is tens of MB against a 32 KB instruction cache on the Vita). With PORT_GAME_HOT (a list of generated file
# names, from tools/hotfiles.py and a PORT_PROF profile), the listed files get PORT_GAME_HOT_OPT and the rest PORT_GAME_OPT,
# which then defaults to -Os: code that rarely runs is made small, code that runs a lot is made fast.
set(PORT_GAME_OPT "" CACHE STRING "optimisation flags for the generated code (e.g. -O2 or -Os); empty: the build type's (-Os with PORT_GAME_HOT)")
set(PORT_GAME_HOT "" CACHE FILEPATH "list of hot generated .c file names (tools/hotfiles.py); they get PORT_GAME_HOT_OPT")
set(PORT_GAME_HOT_OPT "-O2" CACHE STRING "optimisation flags for the files listed in PORT_GAME_HOT")
set(GAME_OPT "${PORT_GAME_OPT}")
if(PORT_GAME_HOT AND NOT GAME_OPT)
  set(GAME_OPT -Os)
endif()
if(GAME_OPT)
  separate_arguments(GAME_OPT)
  target_compile_options(game PRIVATE ${GAME_OPT})
endif()
if(PORT_GAME_HOT)
  file(STRINGS ${PORT_GAME_HOT} HOT_FILES REGEX "^[^#]")
  separate_arguments(HOT_OPT UNIX_COMMAND "${PORT_GAME_HOT_OPT}")
  set(NHOT 0)
  foreach(src ${GEN_SRC})
    get_filename_component(b ${src} NAME)
    if(b IN_LIST HOT_FILES)
      set_source_files_properties(${src} PROPERTIES COMPILE_OPTIONS "${HOT_OPT}")
      math(EXPR NHOT "${NHOT} + 1")
    endif()
  endforeach()
  list(LENGTH GEN_SRC NGEN)
  message(STATUS "generated code: ${NHOT} hot files (${PORT_GAME_HOT_OPT}), ${NGEN} in all; the rest ${GAME_OPT}")
  if(NHOT EQUAL 0)
    message(WARNING "none of the files in ${PORT_GAME_HOT} is among the generated sources: is it from this lift?")
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${PORT_GAME_HOT})
endif()

# ---- the shared platform layer: Win32, the C runtime and DirectX on top of plat.h
file(GLOB CORE_SRC ${CMAKE_CURRENT_SOURCE_DIR}/core/*.c ${CMAKE_CURRENT_SOURCE_DIR}/win32/*.c ${CMAKE_CURRENT_SOURCE_DIR}/crt/*.c ${CMAKE_CURRENT_SOURCE_DIR}/dx/*.c)
option(PORT_VIDEO "built-in cutscene decoder (AVI, Indeo 5, PCM)" ON)
set(VIDEO_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../video)
if(NOT PORT_VIDEO)
  list(REMOVE_ITEM CORE_SRC ${CMAKE_CURRENT_SOURCE_DIR}/dx/video.c)
endif()
if(PORT_ABI_CHECK)
  list(APPEND CORE_SRC ${CMAKE_CURRENT_SOURCE_DIR}/../runtime/rt_abi_check.c)
  set_source_files_properties(${CMAKE_CURRENT_SOURCE_DIR}/../runtime/rt_abi_check.c PROPERTIES COMPILE_DEFINITIONS "RT_ABI_LOG=port_warn")
endif()
add_library(portcore STATIC ${CORE_SRC})
option(PORT_SHADERS "translate the game's pixel shaders to GLSL (dx/d3d9_shader.c); OFF: report none, the game skips its post-processing" ON)
if(NOT PORT_SHADERS)
  target_compile_definitions(portcore PRIVATE PORT_NO_SHADERS)
endif()
option(PORT_PPL "per-pixel lighting for models on by default (dx/d3d9.c, ppl_*; needs PORT_SHADERS); OFF: off unless SS2PORT_PPL=1" OFF)
if(PORT_PPL)
  target_compile_definitions(portcore PRIVATE PORT_PPL_DEFAULT)
endif()
target_include_directories(portcore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/core ${CMAKE_CURRENT_SOURCE_DIR}/../runtime)
target_compile_definitions(portcore PRIVATE ${GEN_DEFS})
target_compile_options(portcore PRIVATE -Wall -Wno-unused-parameter -Wno-misleading-indentation -Wno-format-truncation -Wno-empty-body -fwrapv -fno-strict-aliasing)

# ---- the built-in cutscene decoder (stands in for ffmpeg.dll; video/README.md). FFmpeg's Indeo 5 decoder is LGPL.
if(PORT_VIDEO)
  add_library(lavshim STATIC ${VIDEO_DIR}/lavshim.c ${VIDEO_DIR}/ffmpeg/compat/ffcompat.c ${VIDEO_DIR}/ffmpeg/libavcodec/indeo5.c
              ${VIDEO_DIR}/ffmpeg/libavcodec/ivi.c ${VIDEO_DIR}/ffmpeg/libavcodec/ivi_dsp.c ${VIDEO_DIR}/ffmpeg/libavcodec/vlc.c)
  target_include_directories(lavshim PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../runtime ${VIDEO_DIR} ${VIDEO_DIR}/ffmpeg/compat ${VIDEO_DIR}/ffmpeg/libavcodec)
  target_compile_options(lavshim PRIVATE -O2 -w -fwrapv -fno-strict-aliasing)
  target_include_directories(portcore PRIVATE ${VIDEO_DIR})
  target_compile_definitions(portcore PRIVATE PORT_VIDEO=1)
endif()

# ---- the backend
file(GLOB SYS_SRC ${CMAKE_CURRENT_SOURCE_DIR}/backends/${PORT_SYSTEM}/*.c)
file(GLOB BACKEND_SRC ${CMAKE_CURRENT_SOURCE_DIR}/backends/${PORT_BACKEND}/*.c)
if(NOT BACKEND_SRC)
  message(FATAL_ERROR "no backend '${PORT_BACKEND}' in backends/")
endif()
if(PORT_BACKEND STREQUAL "vita")      # PS Vita: its own system part, the shared GL renderer, VPK packaging (backends/vita/vita.cmake)
  include(${CMAKE_CURRENT_SOURCE_DIR}/backends/vita/vita.cmake)
endif()
add_executable(ss2port ${SYS_SRC} ${BACKEND_SRC} $<TARGET_OBJECTS:game>)
target_include_directories(ss2port PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_definitions(ss2port PRIVATE ${GEN_DEFS})
target_link_libraries(ss2port PRIVATE portcore)
if(NOT WIN32)
  target_link_libraries(ss2port PRIVATE m)
endif()
if(PORT_VIDEO)
  target_link_libraries(ss2port PRIVATE lavshim portcore)
endif()
if(PORT_BACKEND STREQUAL "vita")
  vita_setup_target(ss2port)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(ss2port PRIVATE Threads::Threads)
endif()
if(PORT_BACKEND STREQUAL "sdl2")
  find_package(SDL2 REQUIRED)
  target_link_libraries(ss2port PRIVATE SDL2::SDL2)
  if(WIN32)       # ship SDL2.dll next to the exe
    add_custom_command(TARGET ss2port POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:ss2port>)
  endif()
endif()
